## 没用还是有用

在学习一种程序设计语言的时候，我们通常有两种相辅相成的焦虑：
如何用这门语言做点什么有用的东西，一直在学用这门语言做没用的东西。
所以那个《XXX in Action》系列的书籍才会有卖得那么好……
可是，那个书里依然都是一些有用的没用东西。
到底是有用的没用东西，还是没用的有用东西呢？
实际上，有用的东西多半没用，而没用的东西多半很有用。

    有无相生，
    难易相成。

还有一个就是有用的东西很难学，而没用的东西很容易学。
有些时候，难学的东西居然很没用，而容易学的东西却很有用。

所以，说一千道一万，我们还是先从没用的东西开始学吧。

## 没用的有用项目

我们先来看一个例子：[Checklist Manager](http://8.152.97.118:8090/)

### 一个小游戏：从 API_BASE 拉取随机 checklist

随机显示一个 checklist，里面的每一项都是一个 step，用户可以点击 checkbox 来完成每一项。每一项的 checkbox 只能在前一项完成后才能解锁。

<button id="random-list-btn" type="button" style="padding: 0.5rem 1rem; border: none; background: #2563eb; color: white; border-radius: 0.5rem; cursor: pointer;">加载随机 checklist</button>

<div id="random-list-container" style="margin-top: 1rem;">
  <p>点击按钮后，会从 <code>API_BASE/api/checklists/random</code> 拉取随机清单，并在这里渲染 checkbox。</p>
</div>

<script>
  const API_BASE = 'http://8.152.97.118:8091';
  const RANDOM_CHECKLIST_URL = `${API_BASE}/api/checklists/random`;

  function renderChecklist(checklist) {
    const container = document.getElementById('random-list-container');
    const items = Array.isArray(checklist?.items) ? checklist.items : [];

    const card = document.createElement('div');
    card.style.border = '1px solid #d1d5db';
    card.style.borderRadius = '0.75rem';
    card.style.padding = '1rem 1.25rem';
    card.style.background = '#f9fafb';
    card.style.marginTop = '0.75rem';

    const title = document.createElement('div');
    title.textContent = checklist?.title || '随机 checklist';
    title.style.fontWeight = '700';
    title.style.color = '#111827';
    title.style.fontSize = '1.05rem';
    title.style.marginBottom = '0.35rem';

    const description = document.createElement('div');
    description.textContent = checklist?.description || '暂无说明';
    description.style.color = '#374151';
    description.style.marginBottom = '0.75rem';
    description.style.lineHeight = '1.6';

    const list = document.createElement('ul');
    list.style.listStyle = 'none';
    list.style.paddingLeft = '0';
    list.style.margin = '0';

    items.forEach((item, index) => {
      const li = document.createElement('li');
      li.style.marginBottom = '0.5rem';

      const checkbox = document.createElement('input');
      checkbox.type = 'checkbox';
      checkbox.id = `check-${index}`;
      checkbox.dataset.index = String(index);
      checkbox.style.marginRight = '0.5rem';

      const isUnlocked = index === 0 ||
        Array.from({ length: index }).some((_, prevIndex) => {
          const previousCheckbox = document.getElementById(`check-${prevIndex}`);
          return previousCheckbox && previousCheckbox.checked;
        });

      checkbox.disabled = !isUnlocked;
      checkbox.style.cursor = isUnlocked ? 'pointer' : 'not-allowed';
      checkbox.style.opacity = isUnlocked ? '1' : '0.5';

      checkbox.addEventListener('change', function () {
        const currentIndex = Number(this.dataset.index);

        if (this.checked) {
          for (let i = currentIndex + 1; i < items.length; i += 1) {
            const laterCheckbox = document.getElementById(`check-${i}`);
            if (laterCheckbox) {
              laterCheckbox.disabled = true;
              laterCheckbox.style.opacity = '0.5';
              laterCheckbox.style.cursor = 'not-allowed';
            }
          }

          const nextIndex = currentIndex + 1;
          const nextCheckbox = document.getElementById(`check-${nextIndex}`);
          if (nextCheckbox) {
            nextCheckbox.disabled = false;
            nextCheckbox.style.opacity = '1';
            nextCheckbox.style.cursor = 'pointer';
          }
        } else {
          for (let i = currentIndex + 1; i < items.length; i += 1) {
            const laterCheckbox = document.getElementById(`check-${i}`);
            if (laterCheckbox) {
              laterCheckbox.checked = false;
              laterCheckbox.disabled = true;
              laterCheckbox.style.opacity = '0.5';
              laterCheckbox.style.cursor = 'not-allowed';
            }
          }
        }
      });

      const label = document.createElement('label');
      label.htmlFor = checkbox.id;
      label.textContent = typeof item === 'string' ? item : (item?.step ?? JSON.stringify(item));
      label.style.color = checkbox.disabled ? '#9ca3af' : '#111827';

      li.appendChild(checkbox);
      li.appendChild(label);
      list.appendChild(li);
    });

    card.appendChild(title);
    card.appendChild(description);
    card.appendChild(list);
    container.innerHTML = '';
    container.appendChild(card);
  }

  async function fetchRandomChecklist() {
    const container = document.getElementById('random-list-container');
    container.innerHTML = '<p>正在从 API_BASE 拉取随机 checklist...</p>';

    try {
      const res = await fetch(RANDOM_CHECKLIST_URL, {
        method: 'GET',
        headers: { Accept: 'application/json' }
      });

      if (!res.ok) {
        throw new Error(`HTTP ${res.status}`);
      }

      const payload = await res.json();
      if (!payload || !Array.isArray(payload.items) || payload.items.length === 0) {
        throw new Error('随机接口返回的 checklist 没有 items');
      }

      renderChecklist(payload);
    } catch (err) {
      container.innerHTML = `
        <p>拉取随机 checklist 失败。</p>
        <p>请检查接口是否真的存在：<code>${RANDOM_CHECKLIST_URL}</code></p>
        <p>也可以直接查看 <a href="${API_BASE}/api/docs" target="_blank" rel="noopener noreferrer">OpenAPI 文档</a>。</p>
      `;
    }
  }

  document.getElementById('random-list-btn').addEventListener('click', fetchRandomChecklist);
</script>

OpenAPI 文档：<a href="http://8.152.97.118:8091/api/docs" target="_blank" rel="noopener noreferrer">http://8.152.97.118:8091/api/docs</a>，随机一个checklist的接口是 <code>/api/checklists/random</code>，返回值是一个 checklist 对象，里面的 <code>items</code> 是数组，每一项的字段是 <code>step</code>。
